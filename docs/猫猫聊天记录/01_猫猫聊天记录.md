```python
def _extract_table_chunks(self, blocks, tables):
    """从 PDF artifact 提取 table 为独立 chunk"""
    table_chunks = []
    for table_id, table_data in tables.items():
        # 当前可能被忽略或与周围文本混在一起
        chunk = {
            "content": self._format_table_text(table_data),
            "metadata": {
                "chunk_type": "table",
                "table_id": table_id,
                "source_type": "pdf_table",
                # 关键：确保 embedding 包含 table 上下文
                "heading_path": self._find_table_heading(blocks, table_data),
            }
        }
        table_chunks.append(chunk)
    return table_chunks

def _format_table_text(self, table_data):
    """将 table 结构化数据转为 embedding-friendly 文本"""
    # 当前可能只是 raw HTML，embedding 效果差
    # 改进：表头 + markdown 格式 + caption
    caption = table_data.get("caption", "")
    rows = table_data.get("rows", [])
    
    text_parts = []
    if caption:
        text_parts.append(f"表格：{caption}")
    
    # 转 markdown table（embedding 更友好）
    if rows:
        header = "| " + " | ".join(rows[0]) + " |"
        separator = "|" + "|".join(["---"] * len(rows[0])) + "|"
        body = "\n".join(
            "| " + " | ".join(row) + " |" 
            for row in rows[1:]
        )
        text_parts.extend([header, separator, body])
    
    return "\n".join(text_parts)
```

```python
def split_markdown_content(self, content: str, metadata: dict) -> list[dict]:
    """按 heading 层级切分，保留完整 section"""
    import re
    
    # 1. 解析 heading 结构
    heading_pattern = r'^(#{1,6})\s+(.+)$'
    sections = []
    current_section = {"level": 0, "title": "", "content": []}
    
    for line in content.split('\n'):
        heading_match = re.match(heading_pattern, line)
        if heading_match:
            # 保存前一个 section
            if current_section["content"]:
                sections.append(current_section)
            
            # 开始新 section
            level = len(heading_match.group(1))
            title = heading_match.group(2)
            current_section = {
                "level": level,
                "title": title,
                "content": [line]
            }
        else:
            current_section["content"].append(line)
    
    # 保存最后一个 section
    if current_section["content"]:
        sections.append(current_section)
    
    # 2. 按 section 切 chunk（保持逻辑完整性）
    chunks = []
    for section in sections:
        section_text = '\n'.join(section["content"])
        
        # 小 section 直接成 chunk
        if len(section_text.encode()) <= self.chunk_max_size:
            chunks.append({
                "content": section_text,
                "metadata": {
                    **metadata,
                    "heading_path": section["title"],
                    "section_level": section["level"],
                    "chunk_type": "section_complete"
                }
            })
        else:
            # 大 section 二次切分（保留 heading context）
            sub_chunks = self._split_large_section(
                section_text, 
                section["title"]
            )
            chunks.extend(sub_chunks)
    
    return chunks
```

**实现**：

```python
# app/services/dashscope_rerank_scorer.py
from dashscope import Reranker
from loguru import logger
from app.config import config

class DashScopeRerankScorer:
    """阿里云 GTE Rerank 集成"""
    
    def __init__(self, model: str = "gte-rerank"):
        self.model = model
        self.api_key = config.dashscope_api_key
    
    def score(self, query: str, candidates: list) -> list[float]:
        try:
            # 调用 DashScope Rerank API
            response = Reranker.call(
                model=self.model,
                query=query,
                documents=[c.content for c in candidates],
                top_n=len(candidates),  # 返回全部打分
                return_documents=False,
            )
            
            if response.status_code != 200:
                raise RuntimeError(f"Rerank API 失败: {response.message}")
            
            # 提取分数（保持候选顺序）
            scores = [0.0] * len(candidates)
            for result in response.output.results:
                scores[result.index] = result.relevance_score
            
            logger.debug(f"DashScope rerank 完成, query='{query[:30]}...', scores={scores[:3]}")
            return scores

            raise
```

#### **3.2 Shadow 对比实验**

**目标**：不改变用户返回，后台记录 local vs dashscope rerank 的差异

```python
# evals/knowledge_base/rerank_shadow_comparison.py
import json
from app.services.rerank_service import RerankService
from app.services.dashscope_rerank_scorer import DashScopeRerankScorer
from app.services.rerank_service import LexicalRerankScorer

def run_shadow_comparison(evalset_path: str):
    """对比 local 和 dashscope rerank"""
    
    # 初始化两个 scorer
    local_service = RerankService(
        enabled=True,
        scorer=LexicalRerankScorer()
    )
    dashscope_service = RerankService(
        enabled=True,
        scorer=DashScopeRerankScorer()
    )
    
    results = []
    for sample in load_evalset(evalset_path):
        # 1. 先跑 retrieval 获取候选
        candidates = retrieve_candidates(sample["query"], top_k=10)
        
        # 2. 分别 rerank
        local_reranked = local_service.rerank(sample["query"], candidates)
        dashscope_reranked = dashscope_service.rerank(sample["query"], candidates)
        
        # 3. 对比
        local_top3_ids = [c.id for c in local_reranked[:3]]
        dashscope_top3_ids = [c.id for c in dashscope_reranked[:3]]
        
        rank_diff = len(set(local_top3_ids) ^ set(dashscope_top3_ids))
        
        # 4. 判断是否 lift
        expected_doc_id = sample["expected_doc_id"]
        local_hit = any(expected_doc_id in c.metadata["doc_id"] for c in local_reranked[:3])
        dashscope_hit = any(expected_doc_id in c.metadata["doc_id"] for c in dashscope_reranked[:3])
        
        lift_verdict = "no_change"
        if not local_hit and dashscope_hit:
            lift_verdict = "dashscope_lift"
        elif local_hit and not dashscope_hit:
            lift_verdict = "dashscope_degrade"
        
        results.append({
            "sample_id": sample["sample_id"],
            "query": sample["query"],
            "rank_diff": rank_diff,
            "local_hit": local_hit,
            "dashscope_hit": dashscope_hit,
            "lift_verdict": lift_verdict,
        })
    
    # 5. 汇总
    summary = {
        "total": len(results),
        "dashscope_lift": sum(1 for r in results if r["lift_verdict"] == "dashscope_lift"),
        "dashscope_degrade": sum(1 for r in results if r["lift_verdict"] == "dashscope_degrade"),
        "no_change": sum(1 for r in results if r["lift_verdict"] == "no_change"),
    }
    
    print(json.dumps(summary, indent=2, ensure_ascii=False))
    return results

# 运行
results = run_shadow_comparison("evals/knowledge_base/evalsets/department_rag_100q_v1.jsonl")
```

## 🎭 模式对比：Expert Panel vs 狼人杀

### **Expert Panel + Cross-cat Handoff（结构化协作）**

**适用场景**：
- ✅ 技术决策（架构选型、实现方案、优先级排序）
- ✅ 需要**收敛到唯一方案**的问题
- ✅ 猫们角色明确（架构师/审查员/验证者）
- ✅ 有客观评判标准（性能指标、代码质量、测试覆盖率）

**流程**：
```
1. Expert-panel 发起讨论
   ├─ 独立思考（并行）
   ├─ 观点汇总
   ├─ 辩论/澄清（如有分歧）
   └─ 投票/收敛

2. Cross-cat-handoff 分工执行
   ├─ @砚砚 实现 table probe
   ├─ @小狸 验证 + 审计
   └─ @宪宪 整合 + 文档
```

**优点**：
- 🎯 **目标明确**：必须产出可执行的决策
- 📊 **过程可追溯**：每步都有记录
- ⚡ **效率高**：并行思考 + 结构化收敛
- 🔒 **适合生产决策**：有投票/共识机制

**缺点**：
- 相对严肃，缺少创意碰撞
- 需要明确的决策目标（不适合开放式探索）

- ✅ **开放式探索**（没有标准答案的问题）
